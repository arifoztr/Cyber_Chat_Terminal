/**
 * @file crypto.js
 * @module CryptoModule
 * @description Kriptografi ve Anahtar Yönetimi Modülü (v2 — ECDHE + KDF Ratchet).
 * Web Crypto API kullanarak:
 *   - Kalıcı kimlik anahtarı (Identity Key, P-256 ECDH) — IndexedDB'de saklanır, sadece el sıkışmada kullanılır.
 *   - Oturum bazlı geçici anahtar çifti (Ephemeral Key, ECDHE) — her el sıkışmada yeniden üretilir, PFS sağlar.
 *   - KDF Zinciri (HKDF-SHA256 Ratchet) — her mesajda anahtar rotasyonu yaparak ileriye dönük gizlilik sağlar.
 *   - AES-GCM-256 (12-bayt IV) şifreleme / deşifreleme.
 *   - SHA-256 parmak izi ve Signal tarzı Güvenlik Numarası üretimi.
 *   - Bellek hijyeni: Uint8Array + fill(0) ile anahtar silme, sessionStorage'a sır yazılmaz.
 */

/**
 * Yerel istemcinin kalıcı kimlik ECDH anahtar çifti.

 * @type {CryptoKeyPair|null}

let myEcdhKeyPair = null;

/**

 * @type {JsonWebKey|null}

let myPublicKeyJwk = null;

/**
 * Hedef kullanıcılara göre türetilmiş paylaşılan sırlar (targetUserId → hexSecret).

 * @type {Object.<string, string>}

let derivedSecrets = {};

/**

const _ephemeralKeyPairs = {};

/**

/**

const _skippedMessageKeys = {};

/**
 * Onaltılık (Hex) dizeyi Uint8Array bayt dizisine dönüştürür.
 * @function hexToBytes
 * @param {string} hex - Onaltılık dize.
 * @returns {Uint8Array} Dönüştürülen bayt dizisi.
 */
function hexToBytes(hex) {
    const bytes = new Uint8Array(hex.length / 2);
    for (let i = 0; i < bytes.length; i++) {
        bytes[i] = parseInt(hex.slice(i * 2, i * 2 + 2), 16);
    }
    return bytes;
}

/**
 * Uint8Array bayt dizisini onaltılık (hex) dizeye dönüştürür.
 * @function bytesToHex
 * @param {Uint8Array} bytes - Bayt dizisi.
 * @returns {string} Onaltılık dize.
 */
function bytesToHex(bytes) {
    return Array.from(bytes).map(b => b.toString(16).padStart(2, '0')).join('');
}

/**
 * Onaltılık paylaşılan sırdan AES-GCM CryptoKey nesnesi içe aktarır ve önbelleğe alır.
 * @async
 * @function getAesGcmKey
 * @param {string} hexSecret - 256-bit onaltılık gizli anahtar.
 * @returns {Promise<CryptoKey>} AES-GCM algoritmasında kullanılacak CryptoKey nesnesi.
 */
async function getAesGcmKey(hexSecret) {
    if (_aesKeyCache[hexSecret]) return _aesKeyCache[hexSecret];
    const rawKey = hexToBytes(hexSecret);
    const key = await crypto.subtle.importKey("raw", rawKey, "AES-GCM", false, ["encrypt", "decrypt"]);
    _aesKeyCache[hexSecret] = key;
    return key;
}

/**
 * Uint8Array ham baytlarından AES-GCM CryptoKey nesnesi oluşturur (önbellek kullanmaz).
 * @async
 * @function getAesGcmKeyFromBytes
 * @param {Uint8Array} rawKeyBytes - 256-bit ham anahtar baytları.
 * @returns {Promise<CryptoKey>} AES-GCM CryptoKey nesnesi.
 */
async function getAesGcmKeyFromBytes(rawKeyBytes) {
    return await crypto.subtle.importKey("raw", rawKeyBytes, "AES-GCM", false, ["encrypt", "decrypt"]);
}

/**
 * Bellekteki ham bayt dizisini veya ArrayBuffer'ı Base64 dizesine dönüştürür.
 * Büyük veriler için 8KB'lık parçalar (chunks) kullanarak yığın taşmasını önler.
 * @function bufferToBase64
 * @param {ArrayBuffer|Uint8Array} buffer - Ham veri tamponu.
 * @returns {string} Base64 kodlu dize.
 */
function bufferToBase64(buffer) {
    const bytes = new Uint8Array(buffer);
    const chunks = [];
    for (let i = 0; i < bytes.length; i += 8192) {
        chunks.push(String.fromCharCode(...bytes.subarray(i, i + 8192)));
    }
    const btoaFn = (typeof window !== 'undefined' && window.btoa) ? window.btoa : (typeof btoa !== 'undefined' ? btoa : (b => Buffer.from(b, 'binary').toString('base64')));
    return btoaFn(chunks.join('')); 
}

/**
 * Base64 dizesini Uint8Array bayt dizisine dönüştürür.
 * @function base64ToBuffer
 * @param {string} base64 - Base64 kodlu dize.
 * @returns {Uint8Array} Çözülen bayt dizisi.
 */
function base64ToBuffer(base64) {
    const atobFn = (typeof window !== 'undefined' && window.atob) ? window.atob : (typeof atob !== 'undefined' ? atob : (b => Buffer.from(b, 'base64').toString('binary')));
    const raw = atobFn(base64);
    return Uint8Array.from(raw, c => c.charCodeAt(0));
}

// ============================================================

/**
 * Metni AES-GCM-256 ile şifreler.
 * 12 baytlık rastgele IV üretir ve [IV (12B) + Şifreli Metin (Ciphertext)] şeklinde Base64 olarak paketler.
 * @async
 * @function encryptGCM
 * @param {string} text - Şifrelenecek düz metin.
 * @param {string} hexSecret - 256-bit onaltılık paylaşılan sır.
 * @returns {Promise<string>} Base64 kodlu IV ve şifreli veri paketi.
 */
async function encryptGCM(text, hexSecret) {
    const key = await getAesGcmKey(hexSecret);
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encoded = new TextEncoder().encode(text);
    const ciphertext = await crypto.subtle.encrypt({ name: "AES-GCM", iv: iv }, key, encoded);
    
    const combined = new Uint8Array(iv.length + ciphertext.byteLength);
    combined.set(iv, 0);
    combined.set(new Uint8Array(ciphertext), iv.length);
    
    return bufferToBase64(combined);
}

/**
 * AES-GCM ile şifrelenmiş Base64 veriyi deşifre eder.
 * Verinin ilk 12 baytını IV, geri kalanını şifreli metin olarak ayrıştırır ve UTF-8 metne dönüştürür.
 * @async
 * @function decryptGCM
 * @param {string} base64Data - Base64 formatında [IV + Şifreli Metin].
 * @param {string} hexSecret - 256-bit onaltılık paylaşılan sır.
 * @throws {Error} Veri bozulmuşsa veya anahtar yanlışsa deşifreleme hatası fırlatır.
 * @returns {Promise<string>} Orijinal düz metin.
 */
async function decryptGCM(base64Data, hexSecret) {
    const key = await getAesGcmKey(hexSecret);
    const rawData = base64ToBuffer(base64Data);
    const iv = rawData.slice(0, 12);
    const data = rawData.slice(12);
    
    const decrypted = await crypto.subtle.decrypt({ name: "AES-GCM", iv: iv }, key, data);
    return new TextDecoder().decode(decrypted);
}

/**
 * HKDF-SHA256 ile giriş anahtar materyalinden belirtilen uzunlukta anahtar türetir.
 * @async
 * @function hkdfDerive
 * @param {Uint8Array} ikm - Giriş Anahtar Materyali (Input Keying Material).
 * @param {string} info - Bağlam etiketi (context info).
 * @param {string} [salt='CyberChat-E2EE-v2'] - HKDF tuzu.
 * @param {number} [lengthBits=256] - Türetilecek anahtar uzunluğu (bit).
 * @returns {Promise<Uint8Array>} Türetilmiş anahtar baytları.
 */
async function hkdfDerive(ikm, info, salt = 'CyberChat-E2EE-v2', lengthBits = 256) {
    const hkdfKey = await crypto.subtle.importKey('raw', ikm, 'HKDF', false, ['deriveBits']);
    const saltBytes = new TextEncoder().encode(salt);
    const infoBytes = new TextEncoder().encode(info);
    const derivedBits = await crypto.subtle.deriveBits(
        { name: 'HKDF', hash: 'SHA-256', salt: saltBytes, info: infoBytes },
        hkdfKey,
        lengthBits
    );
    return new Uint8Array(derivedBits);
}

/**
 * Bir hedef kullanıcı ile KDF Ratchet zincirini başlatır.
 * Paylaşılan sırdan (rootKey) deterministik olarak gönderme ve alma zincir anahtarları türetilir.
 * Kullanıcı ID'leri sıralanarak her iki tarafta da aynı sonuç üretilmesi garanti edilir.
 * @async
 * @function initRatchetChain
 * @param {string} targetId - Hedef kullanıcı ID.
 * @param {Uint8Array} sharedSecretBytes - ECDHE'den türetilmiş ham paylaşılan sır.
 * @returns {Promise<void>}
 */
async function initRatchetChain(targetId, sharedSecretBytes) {
    const myId = currentUser.userId;
    // Deterministik sıralama: Her iki tarafta da aynı "chainA" ve "chainB" üretilir
    const [first, second] = [myId, targetId].sort();
    
    const rootKey = await hkdfDerive(sharedSecretBytes, 'ratchet-root-key');
    const chainKeyA = await hkdfDerive(rootKey, 'chain-' + first);
    const chainKeyB = await hkdfDerive(rootKey, 'chain-' + second);

    _ratchetState[targetId] = {
        sendChainKey: (myId === first) ? chainKeyA : chainKeyB,
        recvChainKey: (myId === first) ? chainKeyB : chainKeyA,
        sendCounter: 0,
        recvCounter: 0
    };
    _skippedMessageKeys[targetId] = {};

    // rootKey artık gerekli değil — bellekten sil
    rootKey.fill(0);
}

/**
 * Bir zincir anahtarından mesaj anahtarı türetir ve zinciri bir adım ilerletir.
 * @async
 * @function advanceChain
 * @param {Uint8Array} chainKey - Mevcut zincir anahtarı.
 * @returns {Promise<{ messageKey: Uint8Array, nextChainKey: Uint8Array }>}
 */
async function advanceChain(chainKey) {
    const messageKey = await hkdfDerive(chainKey, 'msg-key');
    const nextChainKey = await hkdfDerive(chainKey, 'chain-advance');
    // Eski zincir anahtarını bellekten sil
    chainKey.fill(0);
    return { messageKey, nextChainKey };
}

/**
 * KDF Ratchet ile mesajı şifreler.
 * Her çağrıda gönderme zinciri bir adım ilerler ve benzersiz bir mesaj anahtarı kullanılır.
 * @async
 * @function encryptWithRatchet
 * @param {string} text - Şifrelenecek düz metin.
 * @param {string} targetId - Hedef kullanıcı ID.
 * @returns {Promise<{ ciphertext: string, messageIndex: number }|null>} Şifreli metin ve mesaj indeksi.
 */
async function encryptWithRatchet(text, targetId) {
    const state = _ratchetState[targetId];
    if (!state) return null;

    const { messageKey, nextChainKey } = await advanceChain(state.sendChainKey);
    state.sendChainKey = nextChainKey;
    const messageIndex = state.sendCounter++;

    try {
        // messageKey (Uint8Array) ile AES-GCM şifreleme
        const aesKey = await getAesGcmKeyFromBytes(messageKey);
        const iv = crypto.getRandomValues(new Uint8Array(12));
        const encoded = new TextEncoder().encode(text);
        const ciphertext = await crypto.subtle.encrypt({ name: "AES-GCM", iv: iv }, aesKey, encoded);

        const combined = new Uint8Array(iv.length + ciphertext.byteLength);
        combined.set(iv, 0);
        combined.set(new Uint8Array(ciphertext), iv.length);

        return { ciphertext: bufferToBase64(combined), messageIndex };
    } finally {
        // Mesaj anahtarını bellekten sil — tek kullanımlık
        messageKey.fill(0);
    }
}

/**
 * KDF Ratchet ile şifreli mesajı çözer.
 * Mesaj indeksine göre alım zincirini ilerletir veya önceden türetilmiş anahtarları kullanır.
 * @async
 * @function decryptWithRatchet
 * @param {string} base64Data - Base64 şifreli veri.
 * @param {string} senderId - Gönderici kullanıcı ID.
 * @param {number} messageIndex - Mesajın zincir indeksi.
 * @returns {Promise<string|null>} Çözülmüş düz metin veya başarısızsa null.
 */
async function decryptWithRatchet(base64Data, senderId, messageIndex) {
    const state = _ratchetState[senderId];
    if (!state) return null;

    let messageKey = null;

    try {
        // 1. Önceden türetilmiş (atlanmış) anahtarlarda kontrol et
        const skipped = _skippedMessageKeys[senderId];
        if (skipped && skipped[messageIndex]) {
            messageKey = skipped[messageIndex];
            delete skipped[messageIndex];
        } else {
            // 2. Alım zincirini mesaj indeksine kadar ilerlet
            while (state.recvCounter < messageIndex) {
                const { messageKey: skippedKey, nextChainKey } = await advanceChain(state.recvChainKey);
                state.recvChainKey = nextChainKey;
                // Atlanan anahtarları sakla (sıra dışı mesajlar için)
                if (!_skippedMessageKeys[senderId]) _skippedMessageKeys[senderId] = {};
                _skippedMessageKeys[senderId][state.recvCounter] = skippedKey;
                state.recvCounter++;
            }
            // 3. Hedef indeksin anahtarını türet
            const result = await advanceChain(state.recvChainKey);
            messageKey = result.messageKey;
            state.recvChainKey = result.nextChainKey;
            state.recvCounter++;
        }

        // AES-GCM deşifreleme
        const aesKey = await getAesGcmKeyFromBytes(messageKey);
        const rawData = base64ToBuffer(base64Data);
        const iv = rawData.slice(0, 12);
        const data = rawData.slice(12);
        const decrypted = await crypto.subtle.decrypt({ name: "AES-GCM", iv: iv }, aesKey, data);
        return new TextDecoder().decode(decrypted);
    } catch (e) {
        console.error('[!] Ratchet deşifreleme hatası:', e);
        return null;
    } finally {
        // Mesaj anahtarını bellekten sil
        if (messageKey) messageKey.fill(0);
    }
}

/**
 * Belirtilen hedef için aktif bir KDF Ratchet zinciri olup olmadığını kontrol eder.
 * @function hasRatchetSession
 * @param {string} targetId - Hedef kullanıcı ID.
 * @returns {boolean}
 */
function hasRatchetSession(targetId) {
    return !!_ratchetState[targetId];
}

/**
 * Belirtilen hedef kullanıcı için oturum bazlı geçici (ephemeral) ECDH anahtar çifti üretir.
 * Özel anahtar extractable: false olarak oluşturulur — JS tarafından dışa aktarılamaz.
 * @async
 * @function generateEphemeralKeyPair
 * @param {string} targetId - Hedef kullanıcı ID.
 * @returns {Promise<JsonWebKey>} Geçici açık anahtar (JWK).
 */
async function generateEphemeralKeyPair(targetId) {
    const keyPair = await crypto.subtle.generateKey(
        { name: 'ECDH', namedCurve: 'P-256' },
        false,  // [GÜVENLİK] extractable: false — özel anahtar JS'den dışa aktarılamaz
        ['deriveBits']
    );
    // Açık anahtarı paylaşmak için JWK'ya dönüştürmemiz gerekiyor,
    // bu yüzden açık anahtarı ayrıca extractable: true olarak import ediyoruz
    const pubJwk = await crypto.subtle.exportKey('jwk', keyPair.publicKey);

    _ephemeralKeyPairs[targetId] = {
        privateKey: keyPair.privateKey,
        publicKey: keyPair.publicKey,
        publicJwk: pubJwk
    };
    return pubJwk;
}

/**
 * Geçici özel anahtarı kullanarak ECDHE ortak sırrı türetir.
 * NIST SP 800-56A: Ham ECDH çıktısı HKDF ile AES anahtarına dönüştürülür.
 * Türetim sonrası geçici özel anahtar referansı bellekten kaldırılır.
 * @async
 * @function deriveEphemeralSecret
 * @param {string} targetId - Hedef kullanıcı ID.
 * @param {JsonWebKey} theirEphemeralPublicJwk - Karşı tarafın geçici açık anahtarı.
 * @returns {Promise<Uint8Array>} 256-bit türetilmiş ortak anahtar (ham baytlar).
 */
async function deriveEphemeralSecret(targetId, theirEphemeralPublicJwk) {
    const ephemeral = _ephemeralKeyPairs[targetId];
    if (!ephemeral) throw new Error('Geçici anahtar bulunamadı: ' + targetId);

    const theirKey = await crypto.subtle.importKey('jwk', theirEphemeralPublicJwk, { name: 'ECDH', namedCurve: 'P-256' }, false, []);
    const rawBits = await crypto.subtle.deriveBits({ name: 'ECDH', public: theirKey }, ephemeral.privateKey, 256);

    // [GÜVENLİK] Geçici özel anahtar artık gerekli değil — referansı kaldır
    // CryptoKey nesnesi GC tarafından toplanacak (JS'de doğrudan sıfırlanamaz)
    delete _ephemeralKeyPairs[targetId].privateKey;

    // HKDF ile ham ECDH çıktısını güçlendir
    const sharedBytes = await hkdfDerive(new Uint8Array(rawBits), 'ECDHE-SharedSecret-v2');
    return sharedBytes;
}

/**
 * Belirtilen hedef için geçici anahtar bilgilerini tamamen temizler.
 * @function wipeEphemeralKey
 * @param {string} targetId - Hedef kullanıcı ID.
 */
function wipeEphemeralKey(targetId) {
    if (_ephemeralKeyPairs[targetId]) {
        delete _ephemeralKeyPairs[targetId];
    }
}

/**
 * Kullanıcının kimlik ECDH anahtar çiftini başlatır.
 * 1. Öncelik: IndexedDB güvenli kasasından CryptoKey olarak yükler.
 * 2. Geriye dönük uyumluluk: localStorage'da eski anahtar varsa IndexedDB'ye taşır ve localStorage'ı temizler.
 * 3. Hiç anahtar yoksa yeni P-256 ECDH çifti oluşturur ve IndexedDB'ye kaydeder.
 * @async
 * @function initEcdhKeys
 * @returns {Promise<void>}
 */
async function initEcdhKeys() {
    // 1. Önce IndexedDB'den CryptoKey nesnesi olarak yüklemeyi dene
    try {
        const vaultKey = await loadCryptoKeyFromVault(currentUser.userId);
        if (vaultKey && vaultKey.privateKey && vaultKey.publicJwk) {
            let pubKey = vaultKey.publicKey;
            if (!pubKey) {
                pubKey = await crypto.subtle.importKey('jwk', vaultKey.publicJwk, { name: 'ECDH', namedCurve: 'P-256' }, true, []);
            }
            myEcdhKeyPair = {
                privateKey: vaultKey.privateKey,
                publicKey: pubKey
            };
            myPublicKeyJwk = vaultKey.publicJwk;
            // Güvenlik: localStorage'da eski anahtar kalmışsa temizle
            localStorage.removeItem('ecdh_keypair_' + currentUser.userId);
            return;
        }
    } catch(e) {
        console.warn('[!] IndexedDB anahtar okuma hatası:', e);
    }

    // 2. Geriye dönük uyumluluk: localStorage'da eski anahtar varsa IndexedDB'ye taşı ve localStorage'dan sil
    const legacyStored = localStorage.getItem('ecdh_keypair_' + currentUser.userId);
    if (legacyStored) {
        try {
            const parsed = JSON.parse(legacyStored);
            const privKey = await crypto.subtle.importKey('jwk', parsed.privateJwk, { name: 'ECDH', namedCurve: 'P-256' }, false, ['deriveBits']);
            const pubKey = await crypto.subtle.importKey('jwk', parsed.publicJwk, { name: 'ECDH', namedCurve: 'P-256' }, true, []);
            
            myEcdhKeyPair = { privateKey: privKey, publicKey: pubKey };
            myPublicKeyJwk = parsed.publicJwk;

            // IndexedDB'ye güvenle kaydet
            await saveCryptoKeyToVault(currentUser.userId, {
                privateKey: privKey,
                publicKey: pubKey,
                publicJwk: myPublicKeyJwk
            });

            // localStorage'dan derhal temizle (Artık düz metin JWK tutulmaz)
            localStorage.removeItem('ecdh_keypair_' + currentUser.userId);
            console.log('[+] Eski ECDH anahtarı IndexedDB kasasına taşındı ve localStorage temizlendi.');
            return;
        } catch(e) {
            console.warn('[!] Eski anahtar taşıma hatası:', e);
        }
    }

    // 3. Hiç anahtar yoksa yeni ECDH çifti oluştur ve doğrudan IndexedDB'ye kaydet
    // [GÜVENLİK] Özel anahtar extractable: false — JS'den dışa aktarılamaz
    // Açık anahtar extractable: true — JWK olarak paylaşılması gerekiyor
    const keyPair = await crypto.subtle.generateKey(
        { name: 'ECDH', namedCurve: 'P-256' },
        false, // extractable: false — XSS saldırısında exportKey ile sızdırılamaz
        ['deriveBits']
    );
    // Açık anahtarı paylaşmak için exportable olarak yeniden import et
    const pubJwk = await crypto.subtle.exportKey('jwk', keyPair.publicKey);
    const exportablePubKey = await crypto.subtle.importKey('jwk', pubJwk, { name: 'ECDH', namedCurve: 'P-256' }, true, []);
    
    myEcdhKeyPair = { privateKey: keyPair.privateKey, publicKey: exportablePubKey };
    myPublicKeyJwk = pubJwk;
    
    await saveCryptoKeyToVault(currentUser.userId, {
        privateKey: myEcdhKeyPair.privateKey,
        publicKey: myEcdhKeyPair.publicKey,
        publicJwk: myPublicKeyJwk
    });

    // Garanti olarak localStorage'da herhangi bir kalıntı olmadığından emin ol
    localStorage.removeItem('ecdh_keypair_' + currentUser.userId);
    console.log('[+] Yeni ECDH kimlik anahtarı IndexedDB kasasında güvenle oluşturuldu (extractable: false).');
}

/**
 * Karşı tarafın JWK açık anahtarı ve yerel özel anahtar ile ECDH ortak sırrı türetir.
 * NIST SP 800-56A standardına uygun olarak ham ortak bitler HKDF fonksiyonundan geçirilerek
 * kriptografik olarak güçlü 256-bit AES anahtarı elde edilir.
 * NOT: Bu fonksiyon geriye dönük uyumluluk için korunmuştur. Yeni el sıkışmalarda
 * deriveEphemeralSecret() tercih edilmelidir.
 * @async
 * @function deriveSharedSecret
 * @param {JsonWebKey} theirPublicJwk - Karşı tarafın JWK açık anahtarı.
 * @returns {Promise<string>} Hex formatında 256-bit türetilmiş ortak anahtar.
 */
async function deriveSharedSecret(theirPublicJwk) {
    const theirKey = await crypto.subtle.importKey('jwk', theirPublicJwk, { name: 'ECDH', namedCurve: 'P-256' }, false, []);
    const rawBits = await crypto.subtle.deriveBits({ name: 'ECDH', public: theirKey }, myEcdhKeyPair.privateKey, 256);
    
    // [GÜVENLİK FIX] NIST SP 800-56A: Ham ECDH çıktısını HKDF ile AES anahtarına dönüştür
    const sharedBytes = await hkdfDerive(new Uint8Array(rawBits), 'AES-GCM-256-SharedKey', 'CyberChat-E2EE-v1');
    const hexResult = bytesToHex(sharedBytes);
    
    // Ham baytları bellekten temizle
    sharedBytes.fill(0);
    
    return hexResult;
}

/**
 * Hedef kullanıcı ile paylaşılan bir simetrik anahtarın var olduğundan emin olur.
 * Bellekte yoksa sunucudan açık anahtar istenir, parmak izi kontrolü (MITM tespiti) yapılır
 * ve ortak anahtar türetilerek saklanır.
 * @async
 * @function ensureSharedSecret
 * @param {string} targetId - Hedef kullanıcının ID'si.
 * @returns {Promise<boolean>} Anahtar başarıyla türetildiyse true, aksi halde false.
 */
async function ensureSharedSecret(targetId) {
    if (derivedSecrets[targetId]) return true;
      
    return new Promise((resolve) => {
        safeEmit('get_public_key', targetId, async (res) => {
            if (res && res.success && res.publicKey) {
                try {
                    // [GÜVENLİK FIX - SEC-05] Açık anahtar parmak izi kontrolü (MITM tespiti)
                    const fingerprint = await generateKeyFingerprint(res.publicKey);
                    const storedFp = localStorage.getItem('cyber_fp_' + currentUser.userId + '_' + targetId);
                    
                    if (storedFp && storedFp !== fingerprint) {
                        // ⚠️ ANAHTAR DEĞİŞMİŞ — Kullanıcıyı uyar
                        console.warn(`[!] GÜVENLİK UYARISI: ${targetId} açık anahtarı değişmiş!`);
                        if (typeof showToast === 'function') {
                            showToast(`⚠️ ${targetId} açık anahtarı değişti! Olası MITM saldırısı. Güvenlik numaranızı doğrulayın.`, 'warning');
                        }
                    }
                    // Parmak izini kaydet/güncelle
                    localStorage.setItem('cyber_fp_' + currentUser.userId + '_' + targetId, fingerprint);
                    
                    const secret = await deriveSharedSecret(res.publicKey);
                    derivedSecrets[targetId] = secret;
                    
                    let contact = myContacts.find(c => c.id === targetId);
                    if (contact) {
                        contact.key = secret;
                        contact.ecdhStatus = 'established';
                        if (res.avatar) contact.avatar = res.avatar;
                        saveContactsToVault();
                        renderContactsSidebarDebounced();
                    }
                    resolve(true);
                } catch(e) { resolve(false); }
            } else {
                resolve(false);
            }
        });
    });
}

/**
 * Hedef kullanıcıya WebSocket üzerinden ECDHE teklifi (offer) göndererek anahtar anlaşmasını başlatır.
 * Yeni mimari: Geçici (ephemeral) anahtar çifti üretilir ve kimlik açık anahtarı ile birlikte gönderilir.
 * @async
 * @function initiateEcdhHandshake
 * @param {string} targetId - Hedef kullanıcının ID'si.
 * @returns {Promise<void>}
 */
async function initiateEcdhHandshake(targetId) {
    if (!myPublicKeyJwk || !socket) return;

    // Oturum bazlı geçici anahtar çifti üret
    const ephemeralPubJwk = await generateEphemeralKeyPair(targetId);

    safeEmit('ecdh_offer', { 
        targetId: targetId, 
        senderId: currentUser.userId, 
        publicKeyJwk: myPublicKeyJwk,           // Kimlik açık anahtarı (parmak izi doğrulama için)
        ephemeralPublicKeyJwk: ephemeralPubJwk   // Geçici açık anahtar (ECDHE için)
    });
}

/**
 * Gelen ECDHE teklifini (offer) işler ve yanıt (answer) hazırlar.
 * Geçici anahtar çifti üretir, ortak sırrı türetir ve KDF ratchet zincirini başlatır.
 * @async
 * @function handleEcdhOffer
 * @param {Object} data - Teklif verisi ({ senderId, publicKeyJwk, ephemeralPublicKeyJwk }).
 * @returns {Promise<{ answerData: Object, secret: string }>} Yanıt verisi ve türetilmiş sır.
 */
async function handleEcdhOffer(data) {
    const hasEphemeral = !!data.ephemeralPublicKeyJwk;

    if (hasEphemeral) {
        // === YENİ ECDHE PROTOKOLÜ ===
        const myEphPubJwk = await generateEphemeralKeyPair(data.senderId);
        const sharedBytes = await deriveEphemeralSecret(data.senderId, data.ephemeralPublicKeyJwk);
        
        // KDF Ratchet zincirini başlat
        await initRatchetChain(data.senderId, sharedBytes);
        
        // Geriye uyumluluk: derivedSecrets'a hex olarak da kaydet
        const hexSecret = bytesToHex(sharedBytes);
        sharedBytes.fill(0); // Ham baytları temizle
        
        wipeEphemeralKey(data.senderId); // Geçici anahtar artık gereksiz

        return {
            answerData: {
                targetId: data.senderId,
                senderId: currentUser.userId,
                publicKeyJwk: myPublicKeyJwk,
                ephemeralPublicKeyJwk: myEphPubJwk
            },
            secret: hexSecret
        };
    } else {
        // === ESKİ PROTOKOL (Geriye Uyumluluk) ===
        const secret = await deriveSharedSecret(data.publicKeyJwk);
        return {
            answerData: {
                targetId: data.senderId,
                senderId: currentUser.userId,
                publicKeyJwk: myPublicKeyJwk
            },
            secret: secret
        };
    }
}

/**
 * Gelen ECDHE yanıtını (answer) işler ve ortak sırrı türetir.
 * @async
 * @function handleEcdhAnswer
 * @param {Object} data - Yanıt verisi ({ senderId, publicKeyJwk, ephemeralPublicKeyJwk }).
 * @returns {Promise<string>} Türetilmiş ortak sır (hex).
 */
async function handleEcdhAnswer(data) {
    const hasEphemeral = !!data.ephemeralPublicKeyJwk;

    if (hasEphemeral && _ephemeralKeyPairs[data.senderId]) {
        // === YENİ ECDHE PROTOKOLÜ ===
        const sharedBytes = await deriveEphemeralSecret(data.senderId, data.ephemeralPublicKeyJwk);
        
        // KDF Ratchet zincirini başlat
        await initRatchetChain(data.senderId, sharedBytes);
        
        const hexSecret = bytesToHex(sharedBytes);
        sharedBytes.fill(0);
        
        wipeEphemeralKey(data.senderId);
        return hexSecret;
    } else {
        // === ESKİ PROTOKOL ===
        return await deriveSharedSecret(data.publicKeyJwk);
    }
}

/**
 * Belirtilen hedef için tüm kriptografik oturum durumunu bellekten güvenle temizler.
 * @function wipeTargetCryptoState
 * @param {string} targetId - Temizlenecek hedef kullanıcı ID.
 */
function wipeTargetCryptoState(targetId) {
    // derivedSecrets temizle
    if (derivedSecrets[targetId]) {
        delete derivedSecrets[targetId];
    }
    // Ratchet zincir anahtarlarını sıfırla
    if (_ratchetState[targetId]) {
        if (_ratchetState[targetId].sendChainKey) _ratchetState[targetId].sendChainKey.fill(0);
        if (_ratchetState[targetId].recvChainKey) _ratchetState[targetId].recvChainKey.fill(0);
        delete _ratchetState[targetId];
    }
    // Atlanmış mesaj anahtarlarını temizle
    if (_skippedMessageKeys[targetId]) {
        Object.values(_skippedMessageKeys[targetId]).forEach(key => { if (key && key.fill) key.fill(0); });
        delete _skippedMessageKeys[targetId];
    }
    // Geçici anahtarları temizle
    wipeEphemeralKey(targetId);
    // AES önbelleğinden ilgili girişleri kaldır
    Object.keys(_aesKeyCache).forEach(k => delete _aesKeyCache[k]);
}

/**
 * Tüm kriptografik durumu bellekten güvenle temizler (oturum kapatma / bağlantı kopması).
 * @function wipeAllCryptoState
 */
function wipeAllCryptoState() {
    // Tüm ratchet zincirlerini sıfırla
    Object.keys(_ratchetState).forEach(id => {
        if (_ratchetState[id].sendChainKey) _ratchetState[id].sendChainKey.fill(0);
        if (_ratchetState[id].recvChainKey) _ratchetState[id].recvChainKey.fill(0);
    });
    Object.keys(_ratchetState).forEach(id => delete _ratchetState[id]);

    // Atlanmış mesaj anahtarlarını sıfırla
    Object.keys(_skippedMessageKeys).forEach(id => {
        Object.values(_skippedMessageKeys[id]).forEach(key => { if (key && key.fill) key.fill(0); });
        delete _skippedMessageKeys[id];
    });

    // Geçici anahtarları temizle
    Object.keys(_ephemeralKeyPairs).forEach(id => delete _ephemeralKeyPairs[id]);

    // AES önbelleğini temizle
    Object.keys(_aesKeyCache).forEach(k => delete _aesKeyCache[k]);

    // Modül değişkenlerini sıfırla
    derivedSecrets = {};
    myEcdhKeyPair = null;
    myPublicKeyJwk = null;
}

/**
 * Bir JWK açık anahtarı için SHA-256 parmak izi üretir.
 * Çıktı: "A1B2 C3D4 E5F6 7890 ..." şeklinde okunabilir 4'lü hex gruplarıdır.
 * @async
 * @function generateKeyFingerprint
 * @param {JsonWebKey} publicKeyJwk - Parmak izi alınacak JWK açık anahtar.
 * @returns {Promise<string|null>} Okunabilir parmak izi dizesi veya anahtar yoksa null.
 */
async function generateKeyFingerprint(publicKeyJwk) {
    if (!publicKeyJwk) return null;
    const keyBytes = new TextEncoder().encode(JSON.stringify(publicKeyJwk));
    const hashBuffer = await crypto.subtle.digest('SHA-256', keyBytes);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const hexStr = hashArray.map(b => b.toString(16).padStart(2, '0').toUpperCase()).join('');
    // 4'lü gruplar halinde formatla (okunabilirlik için)
    return hexStr.match(/.{1,4}/g).join(' ');
}

/**
 * İki kullanıcı arasında Signal protokolü tarzında 60 haneli Güvenlik Numarası (Safety Number) üretir.
 * Her iki tarafın açık anahtarları alfabetik sıralanarak birleştirilir ve SHA-256 ile özetlenir.
 * @async
 * @function generateSafetyNumber
 * @param {JsonWebKey} myPubKeyJwk - Kullanıcının kendi JWK açık anahtarı.
 * @param {JsonWebKey} theirPubKeyJwk - İletişim kurulan karşı tarafın JWK açık anahtarı.
 * @returns {Promise<string|null>} 5'li gruplar halinde formatlanmış güvenlik numarası.
 */
async function generateSafetyNumber(myPubKeyJwk, theirPubKeyJwk) {
    if (!myPubKeyJwk || !theirPubKeyJwk) return null;
    // Sıralama: her iki tarafta da aynı sonucu üretmek için deterministik sırala
    const myStr = JSON.stringify(myPubKeyJwk);
    const theirStr = JSON.stringify(theirPubKeyJwk);
    const combined = [myStr, theirStr].sort().join('|');
    const hashBuffer = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(combined));
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    // İlk 30 baytı 5'li sayısal gruplara dönüştür (60 haneli güvenlik numarası)
    const numStr = hashArray.slice(0, 30).map(b => b.toString(10).padStart(3, '0')).join('');
    return numStr.match(/.{1,5}/g).join(' ');
}

/**
 * Belirtilen kişi için açık anahtar parmak izini sorgular ve önbellekteki ile karşılaştırır.
 * Anahtar değişikliği varsa (olası MITM) kullanıcıyı uyarır.
 * @async
 * @function verifyContactFingerprint
 * @param {string} targetId - Doğrulanacak hedef kişinin ID'si.
 * @returns {Promise<Object>} Doğrulama sonucu ({ verified, fingerprint, firstTime?, previousFingerprint?, reason? }).
 */
async function verifyContactFingerprint(targetId) {
    return new Promise((resolve) => {
        safeEmit('get_public_key', targetId, async (res) => {
            if (!res || !res.success || !res.publicKey) {
                resolve({ verified: false, reason: 'Açık anahtar alınamadı.' });
                return;
            }
            const fingerprint = await generateKeyFingerprint(res.publicKey);
            const storedFp = localStorage.getItem('cyber_fp_' + currentUser.userId + '_' + targetId);
            
            if (!storedFp) {
                // İlk kez görülen anahtar — kaydet
                localStorage.setItem('cyber_fp_' + currentUser.userId + '_' + targetId, fingerprint);
                resolve({ verified: true, fingerprint, firstTime: true });
            } else if (storedFp === fingerprint) {
                // Bilinen anahtar — doğrulandı
                resolve({ verified: true, fingerprint, firstTime: false });
            } else {
                // ⚠️ ANAHTAR DEĞİŞMİŞ — Olası MITM saldırısı!
                resolve({ 
                    verified: false, 
                    fingerprint, 
                    previousFingerprint: storedFp,
                    reason: 'UYARI: Bu kişinin açık anahtarı değişmiş! Olası Ortadaki Adam (MITM) saldırısı.' 
                });
            }
        });
    });
}

    }
}
